document.addEventListener("DOMContentLoaded", () => {
  const questions = document.querySelectorAll(".mayu-question");

  // ------------------------------
  // 各問題の回答処理
  // ------------------------------
  questions.forEach((question) => {
    const buttons = question.querySelectorAll(".mayu-answers button");

    buttons.forEach((button) => {
      button.addEventListener("click", () => {
        buttons.forEach((item) => {
          item.classList.remove("is-selected");
        });

        button.classList.add("is-selected");

        // Q3：答え合わせ
        if (question.dataset.question === "3") {
          const explanation =
            question.querySelector(".mayu-explanation");

          if (explanation) {
            explanation.hidden = false;
          }
        }

        // 通常の補足
        const normalNote =
        question.querySelector(".mayu-note:not(.mayu-answer-note)");

        if (normalNote) {
        normalNote.hidden = false;
        }

        // 回答別の補足
        const answerNotes =
        question.querySelectorAll(".mayu-answer-note");

        if (answerNotes.length > 0) {
        answerNotes.forEach((note) => {
            note.hidden = true;
        });

        const choice = button.dataset.choice;

        const selectedNote =
            question.querySelector(
            `.mayu-answer-note[data-show-answer="${choice}"]`
            );

        if (selectedNote) {
            selectedNote.hidden = false;
        }
        }

        // リラ説明
        const characterNote =
          question.querySelector(".mayu-character-note");

        if (characterNote) {
          characterNote.hidden = false;
        }
      });
    });
  });

  // ------------------------------
  // 最終審査
  // ------------------------------
  const resultButton =
    document.getElementById("mayu-result-button");

  if (!resultButton) {
    return;
  }

  resultButton.addEventListener("click", () => {
    let score = 0;
    let answered = 0;
    let firstUnanswered = null;

    questions.forEach((question) => {
      const selected =
        question.querySelector(
          ".mayu-answers button.is-selected"
        );

      if (selected) {
        answered++;

        const value = Number(selected.dataset.score);

        if (!Number.isNaN(value)) {
          score += value;
        }
      } else if (!firstUnanswered) {
        firstUnanswered = question;
      }
    });

    const resultBox =
      document.getElementById("mayu-result");

    const resultError =
      document.getElementById("mayu-result-error");

    const resultTitle =
      document.getElementById("mayu-result-title");

    const resultMessage =
      document.getElementById("mayu-result-message");

    const resultPoints =
      document.getElementById("mayu-result-points");

    // 未回答あり
    if (answered < questions.length) {
      if (resultError) {
        resultError.textContent =
          `まだ回答していない問題があります。（${answered} / ${questions.length}問回答済み）`;

        resultError.hidden = false;
      }

      if (resultBox) {
        resultBox.hidden = true;
      }

      if (firstUnanswered) {
        firstUnanswered.scrollIntoView({
          behavior: "smooth",
          block: "center"
        });
      }

      return;
    }

    // 必要な結果表示部品があるか確認
    if (
      !resultBox ||
      !resultTitle ||
      !resultMessage ||
      !resultPoints
    ) {
      alert("審査結果の表示欄が見つかりません。");
      return;
    }

    if (resultError) {
      resultError.hidden = true;
    }

    let title = "";
    let message = "";

    if (score <= 7) {
      title = "まゆワールド見学者";

      message = `
        <p>👗「まずは、ゆっくり見学なさってくださいませ。」</p>
        <p>
          まゆワールドには、まだ説明されていないものが
          たくさん存在しております。
          分からないものに出会ったら、
          「これは何？」と聞いてみるところから始めてみましょう。
        </p>
      `;
    } else if (score <= 12) {
      title = "まゆワールド探索者";

      message = `
        <p>👗「なかなか興味深い結果ですわね。」</p>
        <p>
          すでにいくつかの重要な概念を理解し始めていらっしゃいます。
          お兄ちゃんズが一人であることを覚えていただければ、
          今後の探索も多少は楽になるでしょう。
        </p>
      `;
    } else if (score <= 16) {
      title = "まゆワールド理解者";

      message = `
        <p>👗「よろしいでしょう。」</p>
        <p>
          すべてを同じように好きになる必要はありません。
          分からないものでも、
          それが誰かにとって大切なものだと尊重できること。
          それが何より重要ですわ。
        </p>
      `;
    } else {
      title = "リリアーナ公爵令嬢より二次面接推薦";

      message = `
        <p>👗「大変興味深い結果ですわ。」</p>
        <p>
          知らないものを面白がり、
          相手の世界を尊重し、
          必要なときには適切に補助ブレーキを踏める。
        </p>
        <p>
          わたくしから、
          二次面接への推薦を差し上げてもよろしいでしょう。
        </p>
      `;
    }

    resultTitle.textContent = title;
    resultMessage.innerHTML = message;
    resultPoints.textContent = score;

    resultBox.hidden = false;

    resultBox.scrollIntoView({
      behavior: "smooth",
      block: "start"
    });
  });
});
</script>
